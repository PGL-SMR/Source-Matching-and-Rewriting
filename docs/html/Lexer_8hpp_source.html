83private:
87 int getNextChar() {
88 // The current line buffer should not be empty unless it is the end of file.
89 if (CurLineBuffer.empty())
90 return EOF;
91 ++CurCol;
92 auto Nextchar = CurLineBuffer.front();
93 CurLineBuffer = CurLineBuffer.drop_front();
94 if (CurLineBuffer.empty())
95 CurLineBuffer = readNextLine();
96 if (Nextchar == '\n') {
97 ++CurLineNum;
98 CurCol = 0;
99 }
100 return Nextchar;
101 }
102
106 virtual llvm::StringRef readNextLine() = 0;
107
109 Token getTok(bool GetSpace = false) {
110 // Skip any whitespace.
111 while (isspace(LastChar) != 0 && !GetSpace)
112 LastChar = Token(getNextChar());
113
114 // Save the current location before reading the token characters.
115 LastLocation.Line = CurLineNum;
116 LastLocation.Col = CurCol;
117
118 // Check for end of file. Don't eat the EOF.
119 if (LastChar == EOF)
120 return tok_eof;
121
122 // Otherwise, just return the character as its ascii value.
123 auto ThisChar = Token(LastChar);
124 LastChar = Token(getNextChar());
125 return ThisChar;
126 }
127
129 Token CurTok = tok_eof;
130
132 Location LastLocation;
133
135 std::string IdentifierStr;
136
138 double NumVal = 0;
139
143 Token LastChar = Token(' ');
144
146 int CurLineNum = 0;
147
149 int CurCol = 0;
150
152 llvm::StringRef CurLineBuffer = "\n";
153};

156class LexerBuffer final : public Lexer {

158 LexerBuffer(llvm::StringRef Buffer, std::string Filename)
159 : Lexer(std::move(Filename)), Current(Buffer.begin()), End(Buffer.end()) {

161
162private:
165 llvm::StringRef readNextLine() override {
166 const auto *Begin = Current;
167 while (Current <= End && *Current != 0 && *Current != '\n')
168 ++Current;
169 if (Current <= End && *Current != 0)
170 ++Current;
171 llvm::StringRef Result{Begin, static_cast<size_t>(Current - Begin)};
172 return Result;
173 }
174 llvm::StringRef::const_iterator Current;
175 llvm::StringRef::const_iterator End;
176};
